\subsection{Teacher-feature definitions}
\label{app:teacher-feature-definitions}

\paragraph{Common return definitions.}
\begin{align}
\alpha_h
&=
1-\exp\!\left(\frac{\log(0.5)}{h}\right), \\
\operatorname{EWM}_{h,t}(r)
&=
\alpha_h r_t+(1-\alpha_h)\operatorname{EWM}_{h,t-1}(r), \\
r_t^{-}
&=
\min\{r_t,0\}, \\
D_{h,t}(r)
&=
\left[\operatorname{EWM}_{h,t}\!\left((r_t^{-})^2\right)\right]^{1/2}.
\end{align}

\paragraph{Equity teacher features.}
Daily equity excess return: \(r^E_t\). Horizons: \(h\in\{21,63,126,252\}\).
\begin{align}
\operatorname{EWMRet}^{E}_{h,t}
&=
\operatorname{EWM}_{h,t}(r^E), \\
\operatorname{Sortino}^{E}_{h,t}
&=
\frac{\operatorname{EWM}_{h,t}(r^E)}
{D_{h,t}(r^E)+10^{-12}} .
\end{align}

\paragraph{Bond teacher features.}
Monthly bond excess return: \(r^B_t\).
\begin{align}
\operatorname{EWMRet}^{B}_{h,t}
&=
\operatorname{EWM}_{h,t}(r^B),
&& h\in\{3,6,12\}, \\
\operatorname{Sharpe}^{B}_{h,t}
&=
\frac{
\operatorname{mean}(r^B_{t-h+1},\ldots,r^B_t)
}{
\operatorname{sd}(r^B_{t-h+1},\ldots,r^B_t)+10^{-12}
},
&& h\in\{3,6,12\}, \\
\operatorname{LogDownside}^{B}_{h,t}
&=
\log\!\left(D_{h,t}(r^B)+10^{-12}\right),
&& h\in\{1,3,6,12\}, \\
\operatorname{Vol}^{B}_{6,t}
&=
\operatorname{sd}(r^B_{t-5},\ldots,r^B_t).
\end{align}

\subsection{Student base-feature definitions}
\label{app:student-base-feature-definitions}

\paragraph{Rolling scalar features.}
\begin{align}
\operatorname{Sum}_{h,t}(x)
&=
\sum_{j=0}^{h-1}x_{t-j}, \\
\operatorname{Mean}_{h,t}(x)
&=
\frac{1}{h}\sum_{j=0}^{h-1}x_{t-j}, \\
\operatorname{Vol}_{h,t}(x)
&=
\operatorname{sd}(x_{t-h+1},\ldots,x_t).
\end{align}

\paragraph{Return-block features.}
For \(x_{1,t},\ldots,x_{K,t}\),
\begin{align}
\operatorname{Disp}_{t}
&=
\operatorname{sd}(x_{1,t},\ldots,x_{K,t}), \\
\operatorname{BMW}_{t}
&=
\max_{k}x_{k,t}-\min_{k}x_{k,t}.
\end{align}

\paragraph{Correlation-matrix features.}
For the 12-month rolling correlation matrix \(C_t\),
\begin{align}
\overline{\rho}_{t}
&=
\operatorname{mean}_{i<j}(C_{ij,t}), \\
\overline{|\rho|}_{t}
&=
\operatorname{mean}_{i<j}|C_{ij,t}|, \\
\operatorname{PC1}_{t}
&=
\frac{\lambda_{\max}(C_t)}
{\sum_j\lambda_j(C_t)}, \\
\operatorname{Rotation}_{t}
&=
\|C_t-C_{t-1}\|_F .
\end{align}

\paragraph{Fisher correlation changes.}
\begin{align}
\rho_t^{c}
&=
\min\{\max\{\rho_t,-0.999\},0.999\}, \\
F_t
&=
\frac{1}{2}\log\!\left(\frac{1+\rho_t^{c}}{1-\rho_t^{c}}\right), \\
\Delta F_t
&=
F_t-F_{t-1}.
\end{align}

\paragraph{Non-commutativity features.}
\begin{align}
\Delta\Sigma_t
&=
\Sigma_t-\Sigma_{t-1}, \\
\Lambda^{\Sigma}_t
&=
\log\!\left(
1+
100
\frac{
\|\Sigma_t\Delta\Sigma_t-\Delta\Sigma_t\Sigma_t\|_F
}{
\|\Sigma_t\|_F\|\Delta\Sigma_t\|_F
}
\right).
\end{align}
The same formula defines \(\Lambda^{C}_t\) after replacing \(\Sigma_t\) by \(C_t\). Here \(\|\cdot\|_F\) denotes the Frobenius norm.

\paragraph{Differences.}
\begin{align}
\Delta x_t
&=
x_t-x_{t-1}, \\
\Delta\log x_t
&=
\log(x_t)-\log(x_{t-1}).
\end{align}

\subsection{Student-feature transformations}
\label{app:student-feature-transformations}

\paragraph{Causal winsorisation.}
\begin{align}
L_t
&=
Q_{0.01}(x_{t-120},\ldots,x_{t-1}), \\
U_t
&=
Q_{0.99}(x_{t-120},\ldots,x_{t-1}), \\
\widetilde{x}_t
&=
\min\{\max(x_t,L_t),U_t\}.
\end{align}
Window: 120 months. Minimum prior observations: 60.

\paragraph{Causal z-score.}
\begin{equation}
z_t
=
\frac{
\widetilde{x}_t-
\operatorname{mean}(\widetilde{x}_{t-60},\ldots,\widetilde{x}_{t-1})
}{
\operatorname{sd}(\widetilde{x}_{t-60},\ldots,\widetilde{x}_{t-1})+10^{-12}
}.
\end{equation}
Window: 60 months. Minimum prior observations: 36.

\paragraph{Smoothed z-score.}
\begin{equation}
z^s_t
=
\operatorname{EWM}_{12,t}(z).
\end{equation}
Half-life: 12 months.

\paragraph{Student feature vector.}
For base features \(j=1,\ldots,134\),
\begin{equation}
Z_t
=
\bigl(z_{1,t},z^s_{1,t},\ldots,z_{134,t},z^s_{134,t}\bigr).
\end{equation}

\subsection{Coupled approach: return definitions and features}

For half-life \(h\), let \(\mathrm{EWM}_{h}(\cdot)\) denote the exponentially weighted moving
average with that half-life. For an excess-return series \(x_t\), define downside deviation
and the Sortino ratio as
\begin{align}
\mathrm{dd}_{h,t}(x)
&=
\Big[\mathrm{EWM}_{h}\big(\min(x_t,0)^2\big)\Big]^{1/2},
\\
\mathrm{Sortino}_{h,t}(x)
&=
\frac{\mathrm{EWM}_{h}(x)}{\mathrm{dd}_{h,t}(x)+\varepsilon},
\qquad
h\in\{1,2,4,12\}.
\end{align}

With \(m_h=\mathrm{EWM}_{h}(x)\), the coupled teacher uses four additional return-based
features:
\begin{align}
\mathrm{dd\_lvl}_t
&=
\log\!\big(\mathrm{dd}_{6,t}(x)+\varepsilon\big),
&
\mathrm{dd\_diff}_t
&=
\log\!\big(\mathrm{dd}_{2,t}+\varepsilon\big)
-
\log\!\big(\mathrm{dd}_{12,t}+\varepsilon\big),
\\
\mathrm{ma\_cross}_t
&=
\frac{m_{3}-m_{12}}{\mathrm{dd}_{6,t}+\varepsilon},
&
\mathrm{trend}_t
&=
\mathrm{EWM}_{6}\!\left(
\frac{x_t-\mathrm{EWM}_{12}(x)}
{\mathrm{EWM\text{-}sd}_{24}\!\big(x_t-\mathrm{EWM}_{12}(x)\big)+\varepsilon}
\right).
\end{align}

The market-derived block for the coupling logistic~\eqref{eq:wt} uses the trailing
12-month equity--bond correlation \(\rho^{\,\mathrm{eq,bd}}_{12,t}\), together with each
leg's \(\mathrm{trend}_t\) and \(\mathrm{dd\_lvl}_t\).

\subsection{Jump-model objective}

For standardised teacher features \(z_t\), the two-state objective is
\begin{equation}
\min_{s_1,\ldots,s_T,\,c_0,c_1}
\sum_{t=1}^{T} d(z_t,c_{s_t})
+
\lambda\sum_{t=2}^{T}\mathbf{1}\{s_t\neq s_{t-1}\},
\qquad
s_t\in\{0,1\}.
\label{eq:app-jump-objective}
\end{equation}

\subsection{\(L^2\) jump model}

\begin{align}
\mathcal{T}_k
&=
\{t:s_t=k\}, \qquad k\in\{0,1\}, \\
c_k
&=
\frac{1}{|\mathcal{T}_k|}
\sum_{t\in\mathcal{T}_k}z_t, \\
d_2(z_t,c_{s_t})
&=
\frac{1}{2}\|z_t-c_{s_t}\|_2^2.
\end{align}

\subsection{\(L^1\) medoid jump model}

\begin{align}
\widetilde{z}_k
&=
\operatorname{median}\{z_t:t\in\mathcal{T}_k\}, \\
m_k
&=
\arg\min_{z_i:\,i\in\mathcal{T}_k}
\|z_i-\widetilde{z}_k\|_1, \\
d_1(z_t,m_{s_t})
&=
\|z_t-m_{s_t}\|_1
=
\sum_{j=1}^{p}|z_{t,j}-m_{s_t,j}|.
\end{align}

\subsection{Penalty grids and optimisation}

\begin{align}
\lambda^{E}
&\in
\{2,4,8,16,32,64,128\}, \\
\lambda^{B}
&\in
\{0,10^{-5},10^{-4},0.001,0.00390625,0.0078125,0.015625,
0.03125,0.0625,0.125,0.25,0.5,1,2,4\}.
\end{align}

Each fit uses five initialisations and at most 100 coordinate-descent iterations. Conditional on the state centres, the state path is updated by dynamic programming.

\subsection{State ordering}

For fitted raw state \(k\), define
\begin{align}
\widehat{\mu}_k
&=
\operatorname{mean}\{r_t:s_t=k\}, \\
\widehat{\sigma}_k
&=
\operatorname{sd}\{r_t:s_t=k\}, \\
\widehat{\operatorname{SR}}_k
&=
\frac{\widehat{\mu}_k}{\widehat{\sigma}_k}.
\end{align}
The favourable state is
\begin{equation}
g
=
\arg\max_{k\in\{0,1\}}
\widehat{\operatorname{SR}}_k .
\end{equation}
If the Sharpe-ratio comparison is tied,
\begin{equation}
g
=
\arg\max_{k\in\{0,1\}}
\widehat{\mu}_k .
\end{equation}

\subsection{Monthly labels}

For daily equity paths, the monthly favourable-state share is
\begin{equation}
p^{E}_m
=
\frac{1}{N_m}
\sum_{t\in m}
\mathbf{1}\{s_t=g\},
\end{equation}
and the monthly equity label is
\begin{equation}
S^{E}_m
=
\mathbf{1}\{p^{E}_m\geq 0.5\}.
\end{equation}

For monthly bond paths,
\begin{equation}
S^{B}_m
=
\mathbf{1}\{s_m=g\}.
\end{equation}

For asset \(a\in\{E,B\}\), majority-vote labels are
\begin{align}
\bar{S}^{a}_m
&=
\frac{1}{K_a}
\sum_{k=1}^{K_a}S^{a}_{m,k}, \\
S^{a,MV}_m
&=
\mathbf{1}\{\bar{S}^{a}_m\geq 0.5\}.
\end{align}

The one-month-ahead student target is
\begin{equation}
Y^a_{m+1}
=
S^{a,MV}_{m+1}.
\end{equation}

\section{Coupled teacher: estimation details and diagnostics}
\label{app:teacher-coupled}

\subsection*{Objective, coupling logistic, and fitting}
For standardised joint features $(\zeq,\zbd)$ the per-month cost is \eqref{eq:emission} and
the path minimises the total of emission and jump costs by Viterbi. The $L^2$ member uses
Euclidean leg distances with mean centroids; the $L^1$ member uses absolute leg distances
with median centroids (two centroids per leg, indexed by the benign/adverse bit). The
coupling probability $\kappa_t$ in \eqref{eq:wt} is fitted by $L^2$-penalised logistic regression
($C=1$) with balanced class weights and clipped to $[0,1]$. Each fit alternates: (i)
re-estimate centroids and refit the coupling logistic given the path; (ii) update the
path by Viterbi given the centroids and $\kappa_t$. At most ten iterations, a single random restart
plus a warm start from the previous month's path; the lowest-objective solution is kept. Per
leg, the lower-mean state is set to adverse; a stressed-share cap then relabels to normal
the stressed months whose equity moving-average crossover exceeds the $18\%$ quantile of its
trailing window, so only the weakest-trend months remain stressed. After a 132-month
warm-up, each month is refit on the expanding history; $(\lambda,\eta)$ is selected per month
from $\lambda\in\{2,3,4,6\}$ and $\eta\in\{1,2,4,8\}$ to maximise the regime separation score
subject to share, stressed-cap and minimum-run-length constraints (Table~\ref{tab:c-panel}).

\section{Student Prediction Details}
\label{app:prediction-models}
\label{app:student}

\subsection{Targets, features, and validation}

The student models predict the one-month-ahead combined majority labels for the equity regime and the bond regime. The feature vector contains the z-score and smoothed z-score transformation of each monthly base predictor.

For each target and model family, prediction is performed with a rolling validation design. The initial training sample is 96 monthly observations, followed by a 36-month validation window. Candidate hyperparameters are selected on the validation data, and the selected setting is estimated on the full available sample before the forecast month. Validation ranking uses AUC, balanced accuracy, Brier score, net Sharpe ratio, drawdown improvement, volatility reduction, and turnover.

Predicted probabilities are averaged over seeds \(\{101,111,121\}\).

\subsection{Model grids}

The logistic student model uses median imputation, feature standardisation, balanced class weights, and \(L^2\) regularisation. The regularisation grid is
\begin{equation}
C\in\{0.03,0.05,0.10,0.25,0.50,1.00\}.
\end{equation}
The random-forest student model uses median imputation, 500 trees, bootstrap sampling, square-root feature subsampling, and balanced subsample class weights. The grid is
\begin{equation}
(d,\ell)\in
\{(2,10),(2,20),(3,10),(3,20),(4,20)\},
\end{equation}
where \(d\) is tree depth and \(\ell\) is the minimum leaf size.

\section{Allocation Details}
\label{app:allocation}
\label{app:alloc-independent}

\subsection{Probability smoothing and shrinkage}

Let \(p_{i,m}\) be the predicted favourable-regime probability for asset \(i\in\{E,B\}\). Predicted probabilities are used directly or smoothed with exponential half-life
\begin{equation}
h\in\{0,3,6\}.
\end{equation}
For each asset and state, historical next-month excess-return moments are estimated from past observations and shrunk toward pre-prediction unconditional moments:
\begin{equation}
a_{i,s,m}
=
\frac{n_{i,s,m}}{n_{i,s,m}+36},
\end{equation}
\begin{equation}
\tilde{\mu}_{i,s,m}
=
a_{i,s,m}\hat{\mu}_{i,s,m}
+
(1-a_{i,s,m})\mu_{i,0},
\end{equation}
\begin{equation}
\tilde{\sigma}^{2}_{i,s,m}
=
a_{i,s,m}\hat{\sigma}^{2}_{i,s,m}
+
(1-a_{i,s,m})\sigma^{2}_{i,0}.
\end{equation}

\subsection{Probability-to-moment mapping}

The probability-mixture mean and variance are
\begin{equation}
\mu_{i,m}
=
(1-p_{i,m})\tilde{\mu}_{i,0,m}
+
p_{i,m}\tilde{\mu}_{i,1,m},
\end{equation}
\begin{equation}
\sigma^2_{i,m}
=
(1-p_{i,m})(\tilde{\sigma}^2_{i,0,m}+\tilde{\mu}_{i,0,m}^{2})
+
p_{i,m}(\tilde{\sigma}^2_{i,1,m}+\tilde{\mu}_{i,1,m}^{2})
-
\mu_{i,m}^{2}.
\end{equation}
The equity--bond correlation is estimated from expanding historical returns and shrunk toward the pre-prediction correlation using the same shrinkage form. The covariance matrix is then formed from the conditional volatilities and the shrunk equity--bond correlation.

\subsection{Portfolio grid and net returns}

The reported \(L^1\) allocation uses the cash-allowed long-only budget
\[
w_E+w_B\le 1,
\]
with equity and bond weights each capped at one and selected on a 0.02 grid. The reported
allocation grid varies
\[
\gamma\in\{0.25,0.5,1,2,3,5\},
\]
\[
c\in\{0.0005,0.0010,0.0025\},
\]
\[
h\in\{0,3,6\},
\]
and
\[
\lambda_{\mathrm{trade}}\in
\{0,0.0005,0.001,0.0025,0.005,0.0075,0.01,0.015,0.02,0.03,0.05\}.
\]
Here \(h\) is the probability-smoothing half-life and \(\lambda_{\mathrm{trade}}\) is the
turnover penalty in the allocation objective.
Monthly net portfolio return is
\begin{equation}
r^{p}_{m+1}
=
w_{E,m}r_{E,m+1}
+
w_{B,m}r_{B,m+1}
-
c\|w_m-w_{m-1}\|_1.
\end{equation}

The sensitivity checks show how the same allocation rule changes when a single parameter is varied around each selected strategy. Higher risk aversion lowers equity exposure and reduces drawdowns in the defensive and low-turnover specifications. Probability smoothing reduces trading by slowing changes in the forecast probabilities. Higher turnover penalties reduce trading and increase cash weights, which lowers turnover but can reduce the Sharpe ratio when the penalty is too large. The moment-shrinkage sweep varies the amount of shrinkage applied to regime-conditional moments and the stock--bond correlation estimate. Larger \(K\) places more weight on unconditional pre-out-of-sample moments, while smaller \(K\) gives more weight to realised state-conditional samples.

\section{Coupled allocation: details and sensitivity}
\label{app:alloc-coupled}

The investable universe is the equity leg, the bond leg, and cash. Weights are long-only with
a gross cap of $1.0$ ($w_{\mathrm{cash}}\in[0,1]$) and are chosen on a $0.05$ grid. The
\Acoup{} allocator uses annualised regime-conditional moments and an exponentially weighted
stock--bond covariance with a 12-month half-life. This differs from the \Bindep{} allocation
grid and covariance estimator reported in Appendix~\ref{app:alloc-independent}.

mapping uses the probability mixture. Portfolios rebalance monthly and pay $10$~bps on
